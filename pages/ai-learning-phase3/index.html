public class TextSplitter {
    private int chunkSize = 1000;
    private int chunkOverlap = 200;
    
    public List<String> split(String text) {
        List<String> chunks = new ArrayList<>();
        int start = 0;
        int end = Math.min(chunkSize, text.length());
        
        while (start < text.length()) {
            chunks.add(text.substring(start, end));
            start = end - chunkOverlap;
            end = Math.min(start + chunkSize, text.length());
        }
        
        return chunks;
    }
}

# 15.2 优化分片策略

实现智能分片策略：

// 路径：ruoyi-ai-admin/src/main/java/com/ruoyi/ai/utils/IntelligentTextSplitter.java

public class IntelligentTextSplitter {
    private int chunkSize = 1000;
    private int chunkOverlap = 200;
    private double overlapRatio = 0.2; // 重叠比例
    
    public List<String> split(String text) {
        List<String> chunks = new ArrayList<>();
        
        // 1. 按段落分割
        List<String> paragraphs = splitByParagraph(text);
        
        // 2. 合并短段落，分割长段落
        List<String> processedParagraphs = processParagraphs(paragraphs);
        
        // 3. 生成最终分块
        StringBuilder currentChunk = new StringBuilder();
        for (String paragraph : processedParagraphs) {
            if (currentChunk.length() + paragraph.length() <= chunkSize) {
                currentChunk.append(paragraph).append("\n\n");
            } else {
                // 保存当前块
                if (currentChunk.length() > 0) {
                    chunks.add(currentChunk.toString().trim());
                }
                
                // 处理当前段落
                if (paragraph.length() > chunkSize) {
                    // 长段落进一步分割
                    chunks.addAll(splitLongParagraph(paragraph));
                } else {
                    // 开始新块
                    currentChunk = new StringBuilder(paragraph).append("\n\n");
                }
            }
        }
        
        // 保存最后一个块
        if (currentChunk.length() > 0) {
            chunks.add(currentChunk.toString().trim());
        }
        
        return chunks;
    }
    
    private List<String> splitByParagraph(String text) {
        List<String> paragraphs = new ArrayList<>();
        String[] parts = text.split("\\n\\s*\\n");
        for (String part : parts) {
            if (!part.trim().isEmpty()) {
                paragraphs.add(part.trim());
            }
        }
        return paragraphs;
    }
    
    private List<String> processParagraphs(List<String> paragraphs) {
        List<String> result = new ArrayList<>();
        StringBuilder current = new StringBuilder();
        
        for (String paragraph : paragraphs) {
            if (current.length() + paragraph.length() <= chunkSize * 0.8) {
                // 合并短段落
                current.append(paragraph).append("\n\n");
            } else {
                if (current.length() > 0) {
                    result.add(current.toString().trim());
                    current = new StringBuilder();
                }
                result.add(paragraph);
            }
        }
        
        if (current.length() > 0) {
            result.add(current.toString().trim());
        }
        
        return result;
    }
    
    private List<String> splitLongParagraph(String paragraph) {
        List<String> chunks = new ArrayList<>();
        int start = 0;
        int end = Math.min(chunkSize, paragraph.length());
        
        while (start < paragraph.length()) {
            // 尝试在句子边界分割
            int lastSentenceEnd = findLastSentenceEnd(paragraph, start, end);
            if (lastSentenceEnd > start + chunkSize * 0.8) {
                end = lastSentenceEnd;
            }
            
            chunks.add(paragraph.substring(start, end).trim());
            start = end - (int)(chunkSize * overlapRatio);
            end = Math.min(start + chunkSize, paragraph.length());
        }
        
        return chunks;
    }
    
    private int findLastSentenceEnd(String text, int start, int end) {
        int lastPeriod = text.lastIndexOf('.', end);
        int lastExclamation = text.lastIndexOf('!', end);
        int lastQuestion = text.lastIndexOf('?', end);
        
        int maxPos = Math.max(lastPeriod, Math.max(lastExclamation, lastQuestion));
        return maxPos > start ? maxPos + 1 : end;
    }
}

# 15.3 优化向量检索

实现高级检索策略：

// 路径：ruoyi-ai-admin/src/main/java/com/ruoyi/ai/service/impl/KnowledgeServiceImpl.java

@Service
public class KnowledgeServiceImpl implements KnowledgeService {
    
    @Autowired
    private EmbeddingService embeddingService;
    
    @Autowired
    private MilvusService milvusService;
    
    @Override
    public List<String> retrieve(String query, String knowledgeBaseId, int topK) {
        // 1. 生成查询向量
        float[] queryEmbedding = embeddingService.embed(query);
        
        // 2. 基础检索
        List<SearchResult> basicResults = milvusService.search(knowledgeBaseId, queryEmbedding, topK * 2);
        
        // 3. 重排序
        List<SearchResult> rerankedResults = rerankResults(query, basicResults);
        
        // 4. 提取前topK个结果
        List<String> finalResults = new ArrayList<>();
        for (int i = 0; i < Math.min(topK, rerankedResults.size()); i++) {
            finalResults.add(rerankedResults.get(i).getText());
        }
        
        return finalResults;
    }
    
    private List<SearchResult> rerankResults(String query, List<SearchResult> results) {
        // 使用BM25或其他算法进行重排序
        // 这里简化实现，实际项目中可以使用更复杂的算法
        results.sort((a, b) -> {
            double scoreA = calculateRelevanceScore(query, a.getText());
            double scoreB = calculateRelevanceScore(query, b.getText());
            return Double.compare(scoreB, scoreA);
        });
        return results;
    }
    
    private double calculateRelevanceScore(String query, String text) {
        // 简单的相关性计算，实际项目中可以使用更复杂的算法
        int queryLength = query.length();
        int textLength = text.length();
        int matchCount = 0;
        
        // 计算查询词在文本中的出现次数
        for (String word : query.split("\\s+")) {
            if (text.contains(word)) {
                matchCount++;
            }
        }
        
        // 计算相关性得分
        double termFrequency = (double) matchCount / queryLength;
        double inverseDocumentFrequency = Math.log((double) 1000 / (matchCount + 1)); // 假设有1000个文档
        
        return termFrequency * inverseDocumentFrequency;
    }
}

// 路径：ruoyi-ai-admin/src/main/java/com/ruoyi/ai/service/impl/ModelServiceImpl.java

@Service
public class ModelServiceImpl implements ModelService {
    
    private final Map<String, ChatModel> models = new ConcurrentHashMap<>();
    
    @PostConstruct
    public void init() {
        // 初始化各种模型
        models.put("openai", new OpenAIChatModel());
        models.put("qwen", new QwenChatModel());
        models.put("ernie", new ErnieChatModel());
        models.put("doubao", new DoubaoChatModel());
    }
    
    @Override
    public ChatModel getModel(String modelName) {
        ChatModel model = models.get(modelName);
        if (model == null) {
            throw new IllegalArgumentException("Model not found: " + modelName);
        }
        return model;
    }

// 路径：ruoyi-ai-admin/src/main/java/com/ruoyi/ai/service/impl/ChatServiceImpl.java

@Service
public class ChatServiceImpl implements ChatService {
    
    @Override
    public void stream(ChatRequest request, HttpServletResponse response) {
        // 设置响应头
        response.setContentType("text/event-stream");
        response.setCharacterEncoding("UTF-8");
        response.setHeader("Cache-Control", "no-cache");
        response.setHeader("Connection", "keep-alive");
        response.setHeader("X-Accel-Buffering", "no"); // 禁用Nginx缓冲
        
        // 获取模型实例
        ChatModel model = modelService.getModel(request.getModel());
        
        // 构建Prompt
        Prompt prompt = buildPrompt(request);
        
        // 流式调用模型
        try {
            model.stream(prompt, new StreamListener() {
                private final ServletOutputStream outputStream = response.getOutputStream();
                
                @Override
                public void onMessage(String message) {
                    try {
                        // 发送SSE事件
                        String eventData = "data: " + message + "\n\n";
                        outputStream.write(eventData.getBytes(StandardCharsets.UTF_8));
                        outputStream.flush();
                    } catch (IOException e) {
                        // 连接已关闭，停止流式输出
                        Thread.currentThread().interrupt();
                    }
                }
                
                @Override
                public void onComplete() {
                    try {
                        // 发送结束事件
                        outputStream.write("event: complete\ndata: \n\n".getBytes(StandardCharsets.UTF_8));
                        outputStream.flush();
                        outputStream.close();
                    } catch (IOException e) {
                        e.printStackTrace();
                    }
                }
                
                @Override
                public void onError(Exception e) {
                    try {
                        // 发送错误事件
                        String errorData = "event: error\ndata: " + e.getMessage() + "\n\n";
                        outputStream.write(errorData.getBytes(StandardCharsets.UTF_8));
                        outputStream.flush();
                        outputStream.close();
                    } catch (IOException ex) {
                        ex.printStackTrace();
                    }
                }
            });
        } catch (Exception e) {
            e.printStackTrace();
            try {
                response.getWriter().write("data: 发生错误，请重试\n\n");
                response.getWriter().flush();
            } catch (IOException ex) {
                ex.printStackTrace();
            }
        }
    }
}

Agent实现：

// 路径：ruoyi-ai-admin/src/main/java/com/ruoyi/ai/agent/impl/AgentServiceImpl.java

@Service
public class AgentServiceImpl implements AgentService {
    
    @Autowired
    private List<Tool> tools;
    
    @Autowired
    private ChatService chatService;
    
    @Override
    public String execute(String query) {
        // 1. 构建工具列表信息
        String toolsInfo = buildToolsInfo();
        
        // 2. 构建Prompt
        String prompt = "你是一个智能助手，需要根据用户的问题，选择合适的工具来回答。\n" +
            "可用工具：\n" + toolsInfo + "\n" +
            "如果需要使用工具，请按照以下格式输出：\n" +
            "[工具调用]\n" +
            "工具名称：工具名称\n" +
            "参数：{\"参数名\": \"参数值\"}\n" +
            "[/工具调用]\n" +
            "如果不需要使用工具，请直接回答用户问题。\n" +
            "用户问题：" + query;
        
        // 3. 调用模型获取工具调用决策
        ChatRequest request = new ChatRequest();
        request.setMessage(prompt);
        request.setModel("gpt-3.5-turbo");
        String response = chatService.completions(request);
        
        // 4. 解析工具调用
        if (response.contains("[工具调用]")) {
            return handleToolCall(response);
        } else {
            return response;
        }
    }
    
    private String buildToolsInfo() {
        StringBuilder sb = new StringBuilder();
        for (Tool tool : tools) {
            sb.append(tool.getName()).append(": ").append(tool.getDescription()).append("\n");
        }
        return sb.toString();
    }
    
    private String handleToolCall(String response) {
        // 提取工具名称和参数
        Pattern pattern = Pattern.compile("工具名称：(\\w+)\\n参数：(\\{[^}]+\\})");
        Matcher matcher = pattern.matcher(response);
        
        if (matcher.find()) {
            String toolName = matcher.group(1);
            String paramsJson = matcher.group(2);
            
            // 查找对应工具
            Tool tool = tools.stream()
                .filter(t -> t.getName().equals(toolName))
                .findFirst()
                .orElse(null);
            
            if (tool != null) {
                try {
                    // 解析参数
                    Map<String, Object> params = new ObjectMapper().readValue(paramsJson, Map.class);
                    
                    // 执行工具
                    String toolResult = tool.execute(params);
                    
                    // 将工具执行结果发送给模型，获取最终回答
                    String finalPrompt = "工具执行结果：" + toolResult + "\n请根据工具执行结果，回答用户的问题。";
                    ChatRequest request = new ChatRequest();
                    request.setMessage(finalPrompt);
                    request.setModel("gpt-3.5-turbo");
                    return chatService.completions(request);
                } catch (Exception e) {
                    return "工具执行失败：" + e.getMessage();
                }
            } else {
                return "未找到指定工具：" + toolName;
            }
        } else {
            return "工具调用格式错误";
        }
    }
}
