# 29.1 实现对话记录导出功能

后端实现：

// 路径：ruoyi-ai-admin/src/main/java/com/ruoyi/ai/controller/ChatHistoryController.java

@RestController
@RequestMapping("/ai/chat/history")
public class ChatHistoryController {
    
    @Autowired
    private ChatHistoryService chatHistoryService;
    
    // 导出对话记录
    @GetMapping("/export")
    public void export(@RequestParam String sessionId, HttpServletResponse response) {
        try {
            // 设置响应头
            response.setContentType("application/vnd.ms-excel");
            response.setHeader("Content-Disposition", "attachment; filename=chat_history.xlsx");
            
            // 获取对话记录
            List<AiChatHistory> historyList = chatHistoryService.getChatHistory(sessionId);
            
            // 导出到Excel
            chatHistoryService.exportToExcel(historyList, response.getOutputStream());
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

// 路径：ruoyi-ai-admin/src/main/java/com/ruoyi/ai/service/impl/ChatHistoryServiceImpl.java

@Service
public class ChatHistoryServiceImpl implements ChatHistoryService {
    
    @Autowired
    private AiChatHistoryMapper chatHistoryMapper;
    
    @Override
    public void exportToExcel(List<AiChatHistory> historyList, OutputStream outputStream) throws Exception {
        // 创建工作簿
        Workbook workbook = new XSSFWorkbook();
        Sheet sheet = workbook.createSheet("对话记录");
        
        // 创建表头
        Row headerRow = sheet.createRow(0);
        headerRow.createCell(0).setCellValue("序号");
        headerRow.createCell(1).setCellValue("角色");
        headerRow.createCell(2).setCellValue("内容");
        headerRow.createCell(3).setCellValue("时间");
        
        // 填充数据
        for (int i = 0; i < historyList.size(); i++) {
            AiChatHistory history = historyList.get(i);
            Row row = sheet.createRow(i + 1);
            row.createCell(0).setCellValue(i + 1);
            row.createCell(1).setCellValue(history.getRole());
            row.createCell(2).setCellValue(history.getContent());
            row.createCell(3).setCellValue(new SimpleDateFormat("yyyy-MM-dd HH:mm:ss").format(history.getCreateTime()));
        }
        
        // 自动调整列宽
        for (int i = 0; i < 4; i++) {
            sheet.autoSizeColumn(i);
        }
        
        // 写入输出流
        workbook.write(outputStream);
        workbook.close();
    }
}

const sessionId = ref('');

  exportLoading.value = true;

    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = 'chat_history.xlsx';
    link.click();
    URL.revokeObjectURL(url);
  } catch (error) {
    console.error('导出失败:', error);
  } finally {
    exportLoading.value = false;
  }
};
</script>

        <span class="dialog-footer">
          <el-button @click="dialogVisible = false">取消</el-button>

<script setup>
import { ref, onMounted } from 'vue';
import { listKnowledgeBase, createKnowledgeBase, updateKnowledgeBase, deleteKnowledgeBase } from '@/api/knowledge';

const knowledgeList = ref([]);
const dialogVisible = ref(false);
const dialogTitle = ref('新建知识库');
const form = ref({});

onMounted(() => {
  loadKnowledgeList();
});

const loadKnowledgeList = async () => {

    knowledgeList.value = response.data;

const handleCreate = () => {
  form.value = {};
  dialogTitle.value = '新建知识库';

      await updateKnowledgeBase(form.value);

      await createKnowledgeBase(form.value);

    dialogVisible.value = false;
    loadKnowledgeList();

    console.error('操作失败:', error);

const handleDelete = async (id) => {
  try {
    await deleteKnowledgeBase(id);
    loadKnowledgeList();
  } catch (error) {
    console.error('删除失败:', error);
  }
};
</script>

# 29.3 实现用户隔离功能

数据库表设计：

后端实现：

// 路径：ruoyi-ai-admin/src/main/java/com/ruoyi/ai/service/impl/KnowledgeBaseServiceImpl.java

@Service
public class KnowledgeBaseServiceImpl implements KnowledgeBaseService {
    
    @Autowired
    private AiKnowledgeBaseMapper knowledgeBaseMapper;
    
    @Autowired
    private AiUserKnowledgeMapper userKnowledgeMapper;
    
    @Override
    public List<AiKnowledgeBase> listByUserId(Long userId) {
        // 查询用户有权限的知识库
        List<Long> knowledgeIds = userKnowledgeMapper.selectByUserId(userId);
        if (knowledgeIds.isEmpty()) {
            return new ArrayList<>();
        }
        return knowledgeBaseMapper.selectByIds(knowledgeIds);
    }
    
    @Override
    public void addUserPermission(Long userId, Long knowledgeId) {
        AiUserKnowledge userKnowledge = new AiUserKnowledge();
        userKnowledge.setUserId(userId);
        userKnowledge.setKnowledgeId(knowledgeId);
        userKnowledge.setCreateTime(new Date());
        userKnowledgeMapper.insert(userKnowledge);
    }
}

# 36.4 实现大模型限流

使用Redis实现限流：

// 路径：ruoyi-ai-admin/src/main/java/com/ruoyi/ai/service/impl/RateLimitServiceImpl.java

@Service
public class RateLimitServiceImpl implements RateLimitService {
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    @Override
    public boolean isAllowed(String key, int limit, int windowSeconds) {
        String redisKey = "rate_limit:" + key;
        
        // 使用Redis Lua脚本实现限流
        String script = "local current = redis.call('get', KEYS[1]); " +
                      "if current then " +
                      "    if tonumber(current) >= tonumber(ARGV[1]) then " +
                      "        return 0; " +
                      "    else " +
                      "        redis.call('incr', KEYS[1]); " +
                      "        return 1; " +
                      "    end; " +
                      "else " +
                      "    redis.call('set', KEYS[1], 1, 'EX', tonumber(ARGV[2])); " +
                      "    return 1; " +
                      "end;";
        
        DefaultRedisScript<Long> redisScript = new DefaultRedisScript<>(script, Long.class);
        Long result = redisTemplate.execute(redisScript, Collections.singletonList(redisKey), String.valueOf(limit), String.valueOf(windowSeconds));
        
        return result != null && result == 1;
    }
}

// 路径：ruoyi-ai-admin/src/main/java/com/ruoyi/ai/aspect/OperationLogAspect.java

@Aspect
@Component
public class OperationLogAspect {
    
    @Autowired
    private OperationLogService operationLogService;
    
    @Pointcut("@annotation(com.ruoyi.ai.annotation.OperationLog)")
    public void logPointcut() {}
    
    @AfterReturning(pointcut = "logPointcut()", returning = "result")
    public void afterReturning(JoinPoint joinPoint, Object result) {
        recordLog(joinPoint, null);
    }
    
    @AfterThrowing(pointcut = "logPointcut()", throwing = "e")
    public void afterThrowing(JoinPoint joinPoint, Exception e) {
        recordLog(joinPoint, e.getMessage());
    }
    
    private void recordLog(JoinPoint joinPoint, String errorMsg) {
        // 获取用户信息
        Long userId = SecurityUtils.getUserId();
        String username = SecurityUtils.getUsername();
        
        // 获取请求信息
        HttpServletRequest request = ((ServletRequestAttributes) RequestContextHolder.getRequestAttributes()).getRequest();
        String ip = getClientIP(request);
        
        // 获取操作信息
        MethodSignature signature = (MethodSignature) joinPoint.getSignature();
        Method method = signature.getMethod();
        OperationLog annotation = method.getAnnotation(OperationLog.class);
        String operation = annotation.value();
        
        // 记录日志
        AiOperationLog log = new AiOperationLog();
        log.setUserId(userId);
        log.setUsername(username);
        log.setOperation(operation);
        log.setIp(ip);
        log.setCreateTime(new Date());
        if (errorMsg != null) {
            log.setErrorMsg(errorMsg);
        }
        
        operationLogService.save(log);
    }
    
    private String getClientIP(HttpServletRequest request) {
        String ip = request.getHeader("X-Forwarded-For");
        if (ip == null || ip.isEmpty()) {
            ip = request.getRemoteAddr();
        }
        return ip;
    }
}

# 39.4 添加敏感词过滤

敏感词过滤实现：

// 路径：ruoyi-ai-admin/src/main/java/com/ruoyi/ai/utils/SensitiveFilter.java

@Component
public class SensitiveFilter {
    
    private final Set<String> sensitiveWords = new HashSet<>();
    
    @PostConstruct
    public void init() {
        // 加载敏感词列表
        sensitiveWords.add("敏感词1");
        sensitiveWords.add("敏感词2");
        // 可以从数据库或配置文件加载
    }
    
    public String filter(String text) {
        if (text == null || text.isEmpty()) {
            return text;
        }
        
        String filteredText = text;
        for (String word : sensitiveWords) {
            filteredText = filteredText.replaceAll(word, "***");
        }
        
        return filteredText;
    }
    
    public boolean containsSensitiveWords(String text) {
        if (text == null || text.isEmpty()) {
            return false;
        }
        
        for (String word : sensitiveWords) {
            if (text.contains(word)) {
                return true;
            }
        }
        
        return false;
    }
}
